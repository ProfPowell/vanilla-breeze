# Allowlist for scripts/lint-doc-classes.mjs (npm run lint:doc-classes).
#
# Classes used in docs pages or demos that VB intentionally does not define.
# Every entry needs a reason. Prefer fixing the markup: a class VB does not
# define teaches readers to write markup that does nothing.
#
#   name             allowed in every file (`prefix-*` wildcards allowed)
#   path: a b c      allowed only in that repo-relative file
#
# Page-local style findings are keyed dup:name / shadow:name
# (e.g. `site/src/pages/docs/x.html: shadow:lead`).
#
# Pre-existing undefined classes are tracked separately in
# scripts/lint-doc-classes.baseline.json (a ratchet: new ones fail).

# --- Foreign conventions -----------------------------------------------------

# Markdown/highlighter convention: <code class="language-x"> names the code
# language for markdown-viewer, mermaid detection and highlighters.
language-*
# Mermaid's own marker class (<pre class="mermaid">), read by mermaid.js.
mermaid
# Google Translate's opt-out class, shown next to translate="no".
notranslate

# .skip-link is a copy-in recipe whose CSS is on
# /docs/accessibility-utilities/#skip-link, not shipped CSS. Used by many
# page-shell samples and demos, so allowed everywhere.
skip-link
